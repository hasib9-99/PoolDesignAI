        category:"Tile",
        tags:["Stone look","Glass center","6x6"],
        // Use your own hosted thumbnail or get permission from NPT

        nptUrl:"https://www.nptpool.com/tile/elements/",
      },
      {
        id:"veracruz-collection",
        title:"Veracruz Series (overview)",
        category:"Tile",
        tags:["Ceramic","Textured"],
        thumb:"/images/veracruz-overview.jpg",
        nptUrl:"https://www.nptpool.com/tile/veracruz/",
      },
      {
        id:"finish-catalog",
        title:"NPT Pool Finish Catalog",
        category:"Finish",
        tags:["Pebble","Quartz","Polished"],
        thumb:"/images/finish-catalog-2024.jpg",
        nptUrl:"https://www.nptpool.com/catalogs/",
      },
      {
        id:"hardscapes",
        title:"Hardscapes – Pavers & Coping",
        category:"Hardscape",
        tags:["Pavers","Coping"],
        thumb:"/images/hardscapes.jpg",
        nptUrl:"https://www.nptpool.com/",
      }
    ];

    const grid = document.getElementById('grid');
    const q = document.getElementById('q');
    const cat = document.getElementById('cat');

    function render(list){
      grid.innerHTML = "";
      list.forEach(item=>{
        const card = document.createElement('div');
        card.className = 'card';
        card.innerHTML = `
          <img class="thumb" src="${item.thumb}" alt="${item.title}">
          <div class="content">
            <h3 class="title">${item.title}</h3>
            <div class="meta">${item.category}</div>
            <div class="tags">${item.tags.map(t=>`<span class="tag">${t}</span>`).join('')}</div>
          </div>
          <div class="cta">
            <a class="btn secondary" href="${item.nptUrl}" rel="nofollow noopener" target="_blank">View on NPT</a>
          </div>
        `;
        grid.appendChild(card);
      });
    }

    function applyFilters(){
      const term = q.value.toLowerCase().trim();
      const c = cat.value;
      const filtered = DATA.filter(d=>{
        const inCat = !c || d.category===c;
        const inText = !term || (d.title+d.tags.join(' ')).toLowerCase().includes(term);
        return inCat && inText;
      });
      render(filtered);
    }

    q.addEventListener('input', applyFilters);
    cat.addEventListener('change', applyFilters);
    render(DATA);
  </script>
