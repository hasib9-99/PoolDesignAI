1) Frontend (Formspree form)

Use this exactly on the site (replace placeholders as needed). It posts straight to Formspree and shows a success message without leaving the page.

<form id="contact-form" action="https://formspree.io/f/xandgbkq" method="POST">
  <input type="hidden" name="_subject" value="New Inquiry — PoolDesignConsultant.com">
  <input type="hidden" name="_language" value="en">
  <input type="text"   name="name"  placeholder="Your name" required>
  <input type="email"  name="email" placeholder="Your email" required>
  <textarea name="message" placeholder="How can we help?" required></textarea>
  <button type="submit">Send</button>
</form>

<p id="form-status" style="display:none;">Thanks! We received your message.</p>

<script>
  const form = document.getElementById('contact-form');
  const status = document.getElementById('form-status');

  form.addEventListener('submit', async (e) => {
    e.preventDefault();
    const data = new FormData(form);
    const payload = Object.fromEntries(data.entries());

    const r = await fetch(form.action, {
      method: 'POST',
      headers: { 'Accept': 'application/json' },
      body: data
    });

    if (r.ok) {
      status.style.display = 'block';
      form.reset();
    } else {
      alert('Submission failed. Please try again.');
    }
  });
</script>

2) Backend on Replit (Express + SendGrid)

Install

npm i express @sendgrid/mail


server.js

import express from "express";
import sgMail from "@sendgrid/mail";

const app = express();
app.use(express.json({ limit: "1mb" }));
app.use(express.urlencoded({ extended: true }));

// Replit Secrets: SENDGRID_API_KEY, FROM_EMAIL
sgMail.setApiKey(process.env.SENDGRID_API_KEY);

const brand = {
  fromEmail: process.env.FROM_EMAIL, // "kayne@pooldesignconsultant.com"
  fromName: "Pool Design Consultant",
  site: "https://pooldesignconsultant.com",
};

const htmlAutoReply = ({ name }) => `
<!doctype html>
<html>
  <body style="font-family:Arial,sans-serif;line-height:1.6;margin:0;padding:0;">
    <div style="max-width:560px;margin:0 auto;padding:24px;">
      <h2 style="margin:0 0 12px;">Thanks for reaching out${name ? `, ${name}` : ""}!</h2>
      <p>We received your message and will get back to you shortly.</p>
      <p><strong>What happens next?</strong><br/>
      A designer will review your note and reply typically within one business day.</p>
      <hr style="border:none;border-top:1px solid #eee;margin:20px 0;">
      <p style="margin:0;">— Kayne<br/>
      <a href="${brand.site}">PoolDesignConsultant.com</a></p>
    </div>
  </body>
</html>
`;

app.post("/formspree", async (req, res) => {
  // Formspree sends field values under various formats; handle both JSON and form-encoded.
  const body = req.body || {};
  const name = body.name || body.customer_name || "";
  const email = body.email || body.replyTo || body._replyto || "";
  const message = body.message || body.comments || "";

  if (!email) return res.status(400).send("Missing submitter email");

  const notifyKayne = {
    to: "kayne@pooldesignconsultant.com",
    from: { email: brand.fromEmail, name: brand.fromName },
    subject: "New Website Inquiry",
    text: `From: ${name || "Unknown"} <${email}>\n\n${message || ""}`,
  };

  const autoReply = {
    to: email,
    from: { email: brand.fromEmail, name: brand.fromName },
    subject: "We received your message — PoolDesignConsultant.com",
    html: htmlAutoReply({ name }),
    // Optional: bcc Kayne on the auto-reply:
    // bcc: "kayne@pooldesignconsultant.com",
  };

  try {
    await sgMail.send(notifyKayne);
    await sgMail.send(autoReply);
    res.status(200).send("OK");
  } catch (err) {
    console.error(err?.response?.body || err);
    res.status(500).send("SendGrid error");
  }
});

app.get("/", (_, res) => res.send("OK"));
app.listen(3000, () => console.log("Server running on :3000"));


Replit Secrets

SENDGRID_API_KEY → your SendGrid API key

FROM_EMAIL → kayne@pooldesignconsultant.com

Verify kayne@pooldesignconsultant.com as an approved sender.

(Optional) Set up a dedicated link domain for better deliverability.
